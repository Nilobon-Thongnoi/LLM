QUESTION CS-07 (5 POINTS)
----------------------------------------

function loopTillTen() {
    let num = 0;
    let sum = 0;

    while (num <= 10) {
        sum += num;
        num += 2;
    }

    return sum;
}

console.log(loopTillTen());
