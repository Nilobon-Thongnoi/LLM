========================================
REFERENCE SOLUTION 1: IF / ELSE
========================================

function taxCalculate(carCC) {
    let totalTax = 0;

    if (carCC <= 600) {
        totalTax = carCC * 0.5;
    } else if (carCC <= 1800) {
        totalTax = 300 + (carCC - 600) * 1.5;
    } else {
        totalTax = 300 + 1800 + (carCC - 1800) * 4;
    }

    return Math.trunc(totalTax);
}

console.log(taxCalculate(2000));

EXPECTED OUTPUT:
2900

========================================
REFERENCE SOLUTION 2: SWITCH (TRUE)
========================================

function calculateTax(cc) {
    let tax = 0;

    switch (true) {
        case cc <= 600:
            tax = cc * 0.5;
            break;

        case cc <= 1800:
            tax = 300 + (cc - 600) * 1.5;
            break;

        default:
            tax = 2100 + (cc - 1800) * 4;
    }

    return Math.trunc(tax);
}

console.log(calculateTax(2000));

EXPECTED OUTPUT:
2900

========================================
EXPLANATION
========================================

The tax uses progressive tiers.

Tier 1:
The first 600 CC are charged at 0.5 per CC.

Tier 2:
The capacity between 601 and 1800 CC is charged
at 1.5 per CC, plus the tax for the first tier.

Tier 3:
The capacity above 1800 CC is charged at 4 per CC,
plus 300 + 1800 from the previous tiers.

The function accepts engine capacity as a parameter.
The result is converted to an integer.
